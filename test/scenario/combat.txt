# The M2 loop, played through the game app (script/check runs it):
#
#   bin/launch game --no-present --fixed-step 60 --script test/scenario/combat.txt
#
# Fit a module by clicking in the inventory, undock, go to a belt, wait
# for the pirate that turns up there, lock it, orbit and open fire, loot its
# wreck, then lose the ship, fly the capsule home and be given a new one.
#
# covers: Core/Tactical/Combat Core/Tactical/Aftermath Core/Tactical/Pirates Core/Tactical/Gunnery Game/View/Effects
# smoke: yes

wait 2
# (the ship the pilot begins in, whatever it is: what is asked later is whether they are still in it)
lua _G.mine = GameDebug.Hull()
# (sound, which a script does not otherwise get: at no loudness, so that its
# code runs and nobody hears it)
lua GameDebug.Sound(true, 0)
lua assert(GameDebug.Docked(), 'a new game starts docked')
# (and music with it: a station's, while docked)
frames 3
lua assert(GameDebug.Music() == 'docked', 'docked, the music should be the station\'s, not ' .. GameDebug.Music())
lua assert(GameDebug.Fitted() > 0, 'the ship a pilot begins in has nothing fitted')

# In the inventory, the hangar (the third place down its tree), and on
# its second row the shield extender: a right click, and Fit, the first
# thing its menu offers (1920x1080, the window where it is at first).
# (The hangar is set out by kind: lasers, then shields.)
click Left 114 242
frames 3
click Right 274 252
frames 3
lua assert(GameDebug.CanPress('Menu:Fit'), 'a right click on a module in the hangar should offer to fit it')
click Left 304 268
wait 1
lua assert(GameDebug.Fitted('shield.extender.small') == 1, 'clicking Fit did not fit the extender')

# UNDOCK, at the foot of the fitting screen.
click Left 566 700
until GameDebug.InSpace()
frames 3
lua assert(GameDebug.Music() == 'space', 'out of the station the music should be space\'s')

lua GameDebug.TimeScale(4)
lua assert(GameDebug.Order('WarpTo', 'belt'), 'warp to the belt was refused')
until GameDebug.InWarp()
until GameDebug.Idle()
until GameDebug.Pirates() > 0
# (a pirate on the grid is a fight, to the music)
until GameDebug.Music() == 'combat'
lua assert(GameDebug.Pirates() == 1, 'this seed brings one pirate')

# It starts out of targeting range and comes to us.
until GameDebug.Distance('Pirate') < 18000
lua assert(GameDebug.Lock('Pirate'), 'the lock was refused')
until GameDebug.Locked() > 0
lua assert(GameDebug.Order('Orbit', 'Pirate', 1500), 'orbit was refused')
lua assert(GameDebug.Fire() == 2, 'both guns should open fire')

until GameDebug.Count('wreck') > 0
lua assert(GameDebug.Hull() == _G.mine, 'the player lost the fight')
lua assert(GameDebug.Pirates() == 0, 'the pirate is still there')
lua assert(GameDebug.Wallet() == 11000, 'the bounty was not paid')

lua assert(GameDebug.Order('Approach', 'wreck', 500), 'approach was refused')
until GameDebug.Distance('wreck') < 2000
lua assert(GameDebug.Order('Loot', 'wreck'), 'loot was refused')
wait 1
lua assert(GameDebug.Hold('tag.pirate') == 1, 'the pirate tag was not looted')
lua assert(GameDebug.Count('wreck') == 0, 'an emptied wreck should be gone')

# Lose the ship. Insurance pays, and the capsule can get home.
lua GameDebug.SelfDestruct()
wait 0.5
lua assert(GameDebug.Hull() == 'capsule', 'the pilot should be in a capsule')
lua assert(GameDebug.Wallet() == 23000, 'insurance was not paid')
lua assert(GameDebug.Order('Dock', 'station'), 'dock was refused')
until GameDebug.Docked()
wait 0.5
lua assert(GameDebug.Hull() ~= 'capsule', 'docking in a capsule should give a new ship')

lua assert(GameDebug.Sound() > 5, 'hardly a sound was made in all that')
lua assert(GameDebug.Music() == 'docked' and select(3, GameDebug.Music()) >= 4, 'the music should have gone from station to space to fight and back')

echo combat scenario passed
quit
