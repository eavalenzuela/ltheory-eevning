# The right-click menu, in the game app (script/check runs this):
#
#   bin/launch game --no-present --fixed-step 60 --script test/scenario/context.txt
#
# A right click on a row of the overview brings up what can be done
# about it, and a click somewhere else puts it away. Orbit, keep at
# range and warp to each open a second menu of distances: the station is
# orbited at 10 km and then kept at 25, and the belt is warped to at
# 30 km. A distance picked is the one the plain button and the key use
# from then on.
#
# Then the two other menus of the kind. A right click on a plain button
# that gives its order at a distance brings up that order's distances,
# and one picked is the button's: nothing is ordered. And the menu a
# right click on empty space brings up has what there is in the system
# to go to, by kind: a station picked from it is selected, and the menu
# becomes the one for that station.
#
# covers: Game/UI/Context

wait 2
lua assert(GameDebug.Docked(), 'a new game starts docked')
lua GameDebug.Peaceful(true)
until GameDebug.Press('UNDOCK')
until GameDebug.InSpace()
frames 5
lua assert(GameDebug.Order('Stop'))
lua assert(GameDebug.Select('station'))
frames 3
lua assert(GameDebug.CanPress('Orbit 5 km') and GameDebug.CanPress('Keep 10 km'), 'the plain orders begin at 5 and 10 km')

# The overview is listed nearest first, and by another column at a
# click on its head: by name, and at a second click the other way round.
lua assert(GameDebug.Listed() == 'distance' and not select(2, GameDebug.Listed()), 'the overview begins nearest first')
lua _G.nearest = select(3, GameDebug.Listed())
until GameDebug.Press('Overview:sort:name')
frames 3
lua assert(GameDebug.Listed() == 'name' and not select(2, GameDebug.Listed()), 'a click on a head should list by it')
lua assert(select(3, GameDebug.Listed(1)) <= select(3, GameDebug.Listed(2)), 'by name is A to Z')
until GameDebug.Press('Overview:sort:name')
frames 3
lua assert(select(2, GameDebug.Listed()), 'a second click should turn it round')
lua assert(select(3, GameDebug.Listed(1)) >= select(3, GameDebug.Listed(2)), 'the other way round is Z to A')
until GameDebug.Press('Overview:sort:distance')
frames 3
lua assert(select(3, GameDebug.Listed()) == _G.nearest, 'by distance again, the nearest is first')

# A right click on the first row of the overview (1920x1080 window): the
# menu is up, for what that row is, and a click elsewhere puts it away.
click Right 1700 279
frames 3
lua assert(GameDebug.Selected() ~= '', 'a right click should pick the row out')
lua assert(GameDebug.CanPress('Menu:AlignTo'), 'a right click on the overview should bring up the menu')
click Left 500 620
frames 3
lua assert(not GameDebug.CanPress('Menu:AlignTo'), 'a click elsewhere should put the menu away')

# The station: orbit, from its distances, at 10 km. It is on the grid,
# so the menu has no warp in it.
lua assert(GameDebug.Context('station'))
frames 3
lua assert(GameDebug.CanPress('Menu:Orbit') and GameDebug.CanPress('Menu:Dock'))
lua assert(not GameDebug.CanPress('Menu:WarpTo'), 'what is here is not warped to')
until GameDebug.Press('Menu:Orbit')
frames 3
until GameDebug.Press('Menu:Orbit:10000')
until GameDebug.Doing() == 'orbit'
lua assert(GameDebug.Range('Orbit') == 10000)
frames 3
lua assert(not GameDebug.CanPress('Menu:Orbit'), 'the menu should be gone once an order is given')
lua assert(GameDebug.CanPress('Orbit 10 km'), 'the plain button should orbit at the distance picked')

# Kept at 25 km.
lua assert(GameDebug.Context('station'))
frames 3
until GameDebug.Press('Menu:KeepAtRange')
frames 3
until GameDebug.Press('Menu:KeepAtRange:25000')
until GameDebug.Doing() == 'keep'
lua assert(GameDebug.Range('KeepAtRange') == 25000)

# The key orbits at the 10 km picked before.
lua GameDebug.Key('W')
until GameDebug.Doing() == 'orbit'
lua GameDebug.TimeScale(4)
wait 60
lua assert(math.abs(GameDebug.TargetGap() - 10000) < 1500, 'the key should orbit at 10 km, not ' .. GameDebug.TargetGap())

# The belt, which is not here: warped to, at 30 km.
lua assert(GameDebug.Context('belt'))
frames 3
lua assert(GameDebug.CanPress('Menu:WarpTo') and not GameDebug.CanPress('Menu:Orbit'), 'what is not here is warped to, and nothing else')
until GameDebug.Press('Menu:WarpTo')
frames 3
until GameDebug.Press('Menu:WarpTo:30000')
lua GameDebug.TimeScale(8)
until GameDebug.InWarp()
until GameDebug.Idle()
lua assert(math.abs(GameDebug.Distance('belt') - 30000) < 2000, 'the warp should end 30 km short, not ' .. GameDebug.Distance('belt'))

# A right click on the plain Orbit button (in a 1920x1080 window): its
# distances. 50 km is picked, the button says so, and the ship has been
# given no order.
lua GameDebug.TimeScale(1)
lua assert(GameDebug.Order('Stop'))
until GameDebug.Doing() == 'stop'
lua assert(GameDebug.Select('belt'))
frames 3
lua assert(GameDebug.CanPress('Orbit 10 km'))
click Right 1718 89
frames 3
lua assert(select(2, GameDebug.Range('Orbit')) == 'set', 'a right click on the Orbit button should bring up its distances')
lua assert(GameDebug.CanPress('Set:Orbit:50000') and not GameDebug.CanPress('Menu:Orbit'))
until GameDebug.Press('Set:Orbit:50000')
frames 3
lua assert(GameDebug.Range('Orbit') == 50000 and select(2, GameDebug.Range('Orbit')) == '', 'the distance should be set and the menu gone')
lua assert(GameDebug.CanPress('Orbit 50 km'), 'the plain button should say the distance picked')
lua assert(GameDebug.Doing() == 'stop', 'setting a distance orders nothing')

# Empty space: stop, and what there is to go to. The stations come out
# in a list, nearest first, and the first of them picked is selected,
# with its own menu up where this one was.
lua assert(GameDebug.Context())
frames 3
lua assert(select(2, GameDebug.Range('Orbit')) == 'space')
lua assert(GameDebug.CanPress('Space:stop') and GameDebug.CanPress('Space:station') and GameDebug.CanPress('Space:gate'))
lua assert(GameDebug.CanPress('Space:belt') and GameDebug.CanPress('Space:planet'))
until GameDebug.Press('Space:station')
frames 3
lua assert(GameDebug.CanPress('Space:station:1'), 'the stations should be listed')
until GameDebug.Press('Space:station:1')
frames 3
lua assert(select(2, GameDebug.Range('Orbit')) == 'thing', 'a place picked should have its own menu up')
lua assert(GameDebug.CanPress('Menu:WarpTo'), 'a station that is not here is warped to')
until GameDebug.Press('Menu:WarpTo')
frames 3
until GameDebug.Press('Menu:WarpTo:0')
lua GameDebug.TimeScale(8)
until GameDebug.InWarp()
until GameDebug.Idle()
lua assert(GameDebug.Distance('station') < 20000, 'the warp should end at a station')

echo context scenario passed
quit
