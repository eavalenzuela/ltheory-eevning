# The menus, the save slots and the assets window, in the game app
# (script/check runs this):
#
#   bin/launch game --no-present --fixed-step 60 --script test/scenario/menu.txt
#
# A person is given the opening menu and a script is not, so the menu is
# asked for here. Begin a new game from a seed, save it from the menu in
# play, go back to the opening menu, load the save, look the assets
# over, and throw the save away. (A script's saves are kept under
# cache/, apart from anybody's own.)
#
# covers: Game/UI/Menu Game/Settings Game/View/Graphics Core/Persist/Codec

wait 2
lua assert(GameDebug.Menu() == '', 'a script starts in the game, not at the menu')
lua assert(GameDebug.Seed() == 1)
# (Sound switched off with nothing playing, as a game started with the sound off has it: the engine stopped on that once.)
lua GameDebug.Sound(false)
frames 3

# The opening menu: a new game, from a seed.
lua assert(GameDebug.Menu('main') == 'main')
frames 3
lua assert(not GameDebug.CanPress('UNDOCK'), 'nothing of the game is drawn behind a menu')
# Options: the sound's switch, and a slider each for how loud its effects
# and its music are. (A script's run stays silent whatever is set.)
lua assert(select(2, GameDebug.Options()) == 70 and select(3, GameDebug.Options()) == 70, 'effects and music begin at 70')
until GameDebug.Press('Menu:options')
frames 3
lua assert(GameDebug.CanPress('Menu:sound') and GameDebug.CanPress('Effects volume') and GameDebug.CanPress('Music volume'))
lua GameDebug.Enter('Music volume', 35)
frames 3
lua assert(select(3, GameDebug.Options()) == 35 and select(2, GameDebug.Options()) == 70, 'the music slider should set the music and nothing else')
lua GameDebug.Enter('Effects volume', 90)
frames 3
lua assert(select(2, GameDebug.Options()) == 90 and select(3, GameDebug.Options()) == 35, 'the effects slider should set the effects and nothing else')
# (and by hand: the knob of the music slider, where 35 puts it in a
# 1920x1080 window, dragged to the far left)
move 961 535
frames 2
mousedown Left
frames 2
move 872 535
frames 3
mouseup Left
frames 3
lua assert(select(3, GameDebug.Options()) == 0, 'dragging the music knob to the left end should set the music to nothing, not ' .. select(3, GameDebug.Options()))
until GameDebug.Press('Menu:sound')
frames 3
lua assert(GameDebug.Options() == false, 'the switch should turn the sound off')
lua assert(not GameDebug.CanPress('Music volume'), 'with the sound off the sliders are not to be moved')
until GameDebug.Press('Menu:sound')
frames 3
lua assert(GameDebug.Options() == true)
# How the game's window opens: a click goes on to the next way of the list, and round again.
lua assert(GameDebug.WindowChoice() == '', 'at first the window is as the game sees fit')
until GameDebug.Press('Menu:window')
until GameDebug.WindowChoice() == 'borderless'
until GameDebug.Press('Menu:window')
until GameDebug.WindowChoice() == '1920x1080'
# Keys: each thing a key does can be given another. Stopping is Space.
# Its button is clicked, and the next key pressed (X, on the keyboard)
# is its key.
until GameDebug.Press('Menu:keys')
frames 3
lua assert(GameDebug.KeyOf('stop') == 'Space', 'stopping begins on Space')
lua assert(not GameDebug.CanPress('Menu:defaults'), 'with every key as it was at first there is nothing to put back')
until GameDebug.Press('Key:stop')
until select(2, GameDebug.KeyOf('stop')) == 'stop'
key X
frames 3
lua assert(GameDebug.KeyOf('stop') == 'X', 'the key pressed should be the key of what was clicked, not ' .. GameDebug.KeyOf('stop'))
lua assert(select(2, GameDebug.KeyOf('stop')) == '', 'with the key given, no more is listened for')
# A key something else has changes hands: approaching takes W, and
# orbiting gets the Q it gave up.
until GameDebug.Press('Key:approach')
until select(2, GameDebug.KeyOf('approach')) == 'approach'
key W
frames 3
lua assert(GameDebug.KeyOf('approach') == 'W' and GameDebug.KeyOf('orbit') == 'Q', 'the two should have changed keys')
# Escape, while a key is listened for, leaves it as it was, and the menu up.
until GameDebug.Press('Key:map')
until select(2, GameDebug.KeyOf('map')) == 'map'
key Escape
frames 3
lua assert(select(2, GameDebug.KeyOf('map')) == '' and GameDebug.KeyOf('map') == 'M', 'Escape should leave the key as it was')
lua assert(GameDebug.Menu() == 'keys', 'and should not have closed the menu')
# And everything back as it was at first.
until GameDebug.Press('Menu:defaults')
frames 3
lua assert(GameDebug.KeyOf('stop') == 'Space' and GameDebug.KeyOf('approach') == 'Q' and GameDebug.KeyOf('orbit') == 'W')
until GameDebug.Press('Menu:back')
frames 3
lua assert(GameDebug.CanPress('Menu:keys'), 'back from the keys is the options')

# The graphics page: a stage of the picture is switched off and on again, and all go on together.
until GameDebug.Press('Menu:graphics')
until GameDebug.Menu() == 'graphics'
lua assert(GameDebug.Graphics() == '', 'every stage of the picture should be on at first')
lua assert(not GameDebug.CanPress('Menu:allon'), 'with every stage on there is nothing to put back')
until GameDebug.Press('Graphics:bloom')
until GameDebug.Graphics() == 'bloom'
until GameDebug.Press('Graphics:flares')
until GameDebug.Graphics() == 'flares,bloom'
frames 5
until GameDebug.Press('Graphics:bloom')
until GameDebug.Graphics() == 'flares'
until GameDebug.Press('Menu:allon')
until GameDebug.Graphics() == ''
# (the look: three radio buttons, one lit; it is kept with the stages, and All on leaves it be)
lua assert(GameDebug.Look() == 'standard', 'ships are drawn in the standard look at first')
until GameDebug.Press('Graphics:look:cel1')
until GameDebug.Look() == 'cel1'
lua assert(GameDebug.Graphics() == 'look:cel1')
until GameDebug.Press('Graphics:bloom')
until GameDebug.Graphics() == 'bloom,look:cel1'
until GameDebug.Press('Graphics:look:cel2')
until GameDebug.Look() == 'cel2'
until GameDebug.Press('Menu:allon')
until GameDebug.Graphics() == 'look:cel2'
until GameDebug.Press('Graphics:look:standard')
until GameDebug.Graphics() == ''
# (and inside the anti-aliasing: what it shows goes round its three views, and its blend goes the plain way and back)
lua assert(GameDebug.AntiAliasing() == 1 and not select(2, GameDebug.AntiAliasing()))
until GameDebug.Press('Graphics:view')
until GameDebug.AntiAliasing() == 2
until GameDebug.Press('Graphics:view')
until GameDebug.AntiAliasing() == 3
frames 5
until GameDebug.Press('Graphics:view')
until GameDebug.AntiAliasing() == 1
until GameDebug.Press('Graphics:blend')
until select(2, GameDebug.AntiAliasing())
frames 5
until GameDebug.Press('Graphics:blend')
until not select(2, GameDebug.AntiAliasing())
until GameDebug.Press('Menu:back')
frames 3
lua assert(GameDebug.CanPress('Menu:keys'), 'back from the graphics is the options')
until GameDebug.Press('Menu:back')
frames 3
lua assert(GameDebug.CanPress('Menu:options'), 'back from the options is the menu they were opened from')
until GameDebug.Press('Menu:play')
frames 3
lua GameDebug.Enter('Seed', 7)
frames 3
until GameDebug.Press('Menu:start')
until GameDebug.Menu() == ''
lua assert(GameDebug.Seed() == 7, 'the new game is not of the seed asked for')
lua assert(GameDebug.Docked() and GameDebug.Tick() < 100, 'a new game starts docked, at the beginning')
lua assert(GameDebug.Hangar('ammo.emp') == 600, 'a new game has the starting kit')

# Into space, and a save from the menu in play, which stops the clock.
until GameDebug.Press('UNDOCK')
until GameDebug.InSpace()
wait 2
lua assert(GameDebug.Menu('pause') == 'pause')
frames 3
lua _G.tick = GameDebug.Tick()
until GameDebug.Press('Menu:save')
frames 3
until GameDebug.Press('Save:save 5')
frames 3
lua assert(GameDebug.CanPress('Save:save 5'), 'the slot should be there to write over')
until GameDebug.Press('Menu:back')
frames 30
lua assert(GameDebug.Tick() == _G.tick, 'the game went on behind the menu')
until GameDebug.Press('Menu:resume')
until GameDebug.Menu() == ''
wait 3
lua assert(GameDebug.Tick() > _G.tick, 'the game did not go on after the menu')

# Back at the opening menu the save can be gone on from, or loaded by name.
lua assert(GameDebug.Menu('main') == 'main')
frames 3
lua assert(GameDebug.CanPress('Menu:continue'), 'with a save there is something to continue')
until GameDebug.Press('Menu:play')
frames 3
until GameDebug.Press('Load:save 5')
until GameDebug.Menu() == ''
lua assert(GameDebug.Seed() == 7 and GameDebug.InSpace(), 'the save did not bring the game back')
lua assert(GameDebug.Tick() >= _G.tick and GameDebug.Tick() < _G.tick + 3, 'the save is not from when it was made')

# What the pilot owns, from anywhere: here, the hangar left behind.
# (it is opened from the inventory, which in space is shut until it is asked for)
until GameDebug.Press('Inventory')
until GameDebug.Press('Assets')
frames 3
lua assert(GameDebug.CanPress('assets:close'), 'the assets window did not open')
until GameDebug.Press('assets:close')
until not GameDebug.CanPress('assets:close')
until GameDebug.Press('Colonies')
frames 3
lua assert(GameDebug.CanPress('Colony:close'), 'the colonies window did not open')
until GameDebug.Press('Colony:close')

# And the save is thrown away.
lua assert(GameDebug.Menu('pause') == 'pause')
frames 3
until GameDebug.Press('Menu:load')
frames 3
until GameDebug.Press('Delete:save 5')
frames 3
lua assert(GameDebug.CanPress('Load:save 5'), 'a save is thrown away only when asked twice')
until GameDebug.Press('Delete:save 5')
frames 3
lua assert(not GameDebug.CanPress('Load:save 5'), 'the deleted save is still listed')

echo menu scenario passed
quit
