# A starbase of the pilot's own, put up through the game app's own
# screens (script/check runs this):
#
#   bin/launch game --no-present --fixed-step 60 --script test/scenario/starbase.txt
#
# Buy a control tower, fuel and a refining array, carry them to a moon
# in an industrial, anchor the tower there, moor at it, fuel it, anchor
# the array, and refine ore at it: a station where there was none, that
# does what the pilot built it to do. Then the tower is taken down
# again, into the hold. The industrial and the price of it all are
# granted. The rest is done as a player does it.
#
# covers: Core/Assets/Starbases Core/Data/Structures Game/View/Starbase Game/UI/Industry Game/UI/Starbase
# smoke: yes

wait 2
lua GameDebug.Peaceful(true)
lua assert(GameDebug.Docked(), 'a new game starts docked')
lua assert(GameDebug.Fly('industrial.mizumushi'), 'the Mizumushi could not be flown')
lua GameDebug.Grant(5000000)
wait 1.5

# The market sells what a starbase is made of, under Structures.
lua assert(GameDebug.Market('tower.vessari.small'))
frames 3
until GameDebug.Press('BuyNow')
until GameDebug.Hangar('tower.vessari.small') == 1
lua assert(GameDebug.Market('array.refining'))
frames 3
until GameDebug.Press('BuyNow')
until GameDebug.Hangar('array.refining') == 1
lua assert(GameDebug.Market('fuel.block'))
frames 3
lua GameDebug.Enter('Quantity', 100)
frames 2
until GameDebug.Press('BuyNow')
until GameDebug.Hangar('fuel.block') == 100
lua assert(GameDebug.Market('ore.ferrite'))
frames 3
lua GameDebug.Enter('Quantity', 300)
frames 2
until GameDebug.Press('BuyNow')
until GameDebug.Hangar('ore.ferrite') == 300

# A pilot cannot anchor anything here: this station is not theirs.
lua assert(GameDebug.Tab() ~= 'Starbase', 'the starter station is not the pilot\'s to run')

# All of it into the hold.
until GameDebug.Press('Fitting')
frames 3
until GameDebug.Press('ToHold:tower.vessari.small')
until GameDebug.Hold('tower.vessari.small') == 1
until GameDebug.Press('ToHold:array.refining')
until GameDebug.Hold('array.refining') == 1
until GameDebug.Press('ToHold:fuel.block')
until GameDebug.Hold('fuel.block') == 100
until GameDebug.Press('ToHold:ore.ferrite')
until GameDebug.Hold('ore.ferrite') == 300

# To a moon. (The nearest may be the one the station stands by, too near
# to warp to: then the pilot is beside it already.)
until GameDebug.Press('UNDOCK')
until GameDebug.InSpace()
lua GameDebug.TimeScale(30)
lua _G.warped = GameDebug.Order('WarpTo', 'moon')
until not _G.warped or GameDebug.InWarp()
until GameDebug.Idle()
lua GameDebug.TimeScale(1)
lua _G.stations = GameDebug.Count('station')

# Anchor the tower there.
lua assert(GameDebug.Select('moon'), 'no moon to select')
frames 3
lua assert(GameDebug.Tower() == 'none')
until GameDebug.Press('Anchor tower')
until GameDebug.Tower() == 'anchoring'
lua assert(GameDebug.Hold('tower.vessari.small') == 0, 'the tower is still in the hold')
lua assert(GameDebug.Count('station') == _G.stations + 1, 'the tower is not on the overview')
lua assert(GameDebug.Distance('Starbase') == 0, 'the pilot should be inside the field of the tower just anchored')

# Nobody moors until it is anchored.
lua assert(GameDebug.Select('Starbase'))
frames 3
lua assert(not GameDebug.CanPress('Dock'), 'a tower still anchoring should not be moored at')
lua GameDebug.TimeScale(10)
until GameDebug.Tower() == 'dark'
lua GameDebug.TimeScale(1)
lua assert(GameDebug.Order('Dock', 'Starbase'), 'the pilot\'s own tower refused them')
until GameDebug.Moored()

# The running of it is what opens. With no fuel it is dark, and nothing
# can be anchored.
frames 3
lua assert(GameDebug.Tab() == 'Starbase', 'the Starbase tab should open at a tower of one\'s own')
lua assert(not GameDebug.CanPress('Anchor:array.refining'), 'a dark tower should take no module')
until GameDebug.Press('Fuel:hold')
until GameDebug.Tower() == 'lit'
lua assert(select(2, GameDebug.Tower()) == 100)

# The array, which is online a little later.
until GameDebug.Press('Anchor:array.refining')
until GameDebug.Hold('array.refining') == 0
lua GameDebug.TimeScale(10)
until select(3, GameDebug.Tower()) == 1
lua GameDebug.TimeScale(1)

# And it refines, as the one at the starter station does.
until GameDebug.Press('Fitting')
frames 3
until GameDebug.Press('ToHangar:ore.ferrite')
until GameDebug.Hangar('ore.ferrite') == 300
until GameDebug.Press('Refine:ore.ferrite')
until GameDebug.Hangar('mineral.iron') > 0

# The array comes off again, into the hangar.
until GameDebug.Press('Starbase')
frames 3
until GameDebug.Press('Unanchor:array.refining')
until GameDebug.Hangar('array.refining') == 1
lua assert(select(3, GameDebug.Tower()) == 0)

# And down it comes again, into the hold: not with anything in its
# hangar (pressed then, it says why not), and with the hangar empty the
# tower is in the hold, the ship in space where the tower was, and
# nothing of it left on the overview.
until GameDebug.Press('Tower:down')
frames 5
lua assert(GameDebug.Tower() ~= 'none' and GameDebug.Moored(), 'a tower with things in its hangar should not come down')
until GameDebug.Press('Fitting')
frames 3
until GameDebug.Press('ToHold:array.refining')
until GameDebug.Hold('array.refining') == 1
lua assert(GameDebug.Lose('all') > 0, 'the minerals should have been in the hangar')
until GameDebug.Press('Starbase')
frames 3
until GameDebug.Press('Tower:down')
until GameDebug.Tower() == 'none'
until GameDebug.InSpace()
frames 5
lua assert(GameDebug.Hold('tower.vessari.small') == 1, 'the tower should be in the hold')
lua assert(GameDebug.Hold('fuel.block') > 0, 'and what fuel was in its tank')
lua assert(not GameDebug.Select('Starbase'), 'the tower should be off the overview')

echo starbase scenario passed
quit
