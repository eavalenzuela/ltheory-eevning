# A galaxy of forty systems, the star map and the autopilot, in the game
# app (script/check runs this):
#
#   bin/launch game --no-present --fixed-step 60 --script test/scenario/galaxy.txt
#
# Begin a game in a galaxy of forty. A pilot starts at the first
# faction's homeworld. Open the map, set the farthest system that can be
# reached without entering lawless space as the destination, by the
# safest way, switch the autopilot on, and be flown there: out of the
# station, gate after gate, with every system on the way built and
# thrown away again.

wait 2
lua GameDebug.NewGame(5, 40)
frames 5
lua GameDebug.Peaceful(true)
lua assert(GameDebug.Galaxy() == 40, 'the galaxy should have forty systems')
lua assert(select(2, GameDebug.Galaxy()) == 'Vessari', 'a pilot starts in the first faction\'s space')
lua assert(GameDebug.Security() == 1, 'a pilot starts at a homeworld')
lua assert(GameDebug.Docked())

# The map, and somewhere far to go.
until GameDebug.Press('Map')
frames 3
lua assert(GameDebug.CanPress('Map:close'), 'the map did not open')
lua assert(not GameDebug.CanPress('Autopilot'), 'with nowhere to go there is no autopilot to switch on')

# The map is looked at nearer and further: by its buttons, by the wheel
# (about what the cursor is on), and dragged about with the left button.
lua assert(GameDebug.MapZoom() == 1 and select(2, GameDebug.MapZoom()) == 40, 'the map should open on all forty systems')
until GameDebug.Press('Map:here')
frames 60
lua _G.near, _G.sighted, _G.named = GameDebug.MapZoom()
lua assert(_G.near == 2 and _G.sighted < 40 and _G.named, 'Here should look twice as near, at fewer systems, each with its name')
lua _G.mx, _G.my = GameDebug.MapWhere(GameDebug.System())
lua assert(math.abs(_G.mx - 810) < 2 and math.abs(_G.my - 550) < 2, 'the system the pilot is in should be in the middle of the map')
move 810 550
frames 2
scroll 2
frames 60
lua assert(math.abs(GameDebug.MapZoom() - 3.125) < 0.01, 'two notches of the wheel should near it by a quarter twice')
lua _G.mx, _G.my = GameDebug.MapWhere(GameDebug.System())
lua assert(math.abs(_G.mx - 810) < 2 and math.abs(_G.my - 550) < 2, 'what the cursor is on should stay under it')
mousedown Left
frames 2
move 910 600
frames 3
mouseup Left
frames 3
lua _G.mx, _G.my = GameDebug.MapWhere(GameDebug.System())
lua assert(math.abs(_G.mx - 910) < 2 and math.abs(_G.my - 600) < 2, 'the left button held should drag the map')
lua assert(GameDebug.MapPicked() == 0, 'a drag is not a click')
click Left 910 600
frames 3
lua assert(GameDebug.MapPicked() == GameDebug.System(), 'a click on a system should pick it')
until GameDebug.Press('Map:fit')
frames 80
lua assert(GameDebug.MapZoom() == 1 and select(2, GameDebug.MapZoom()) == 40, 'Fit should show all the galaxy again')
lua _G.far, _G.jumps, _G.id = GameDebug.Far()
lua print('going to', _G.far, 'which is', _G.jumps, 'jumps off')
lua assert(_G.jumps >= 4, 'a galaxy of forty should have somewhere four jumps off that is not lawless')
lua assert(GameDebug.Destination(_G.far, true) == _G.jumps)
frames 3
until GameDebug.Press('Map:autopilot')
until GameDebug.Autopilot()
until GameDebug.Press('Map:close')

# The autopilot undocks and flies the route. It never enters lawless space.
lua GameDebug.TimeScale(16)
until GameDebug.InSpace()
patience 400
until GameDebug.System() ~= 1
lua assert(select(2, GameDebug.Security()) ~= 'lawless', 'the safest way went through lawless space')
lua assert(GameDebug.Destination() == _G.jumps - 1, 'a jump on, the route should be a jump shorter')
lua assert(GameDebug.GatesLit() >= 1, 'a gate should light when the ship goes through it')
# (a jump is seen: the view whites out over the system left, and clears
# in the next)
frames 8
lua assert(GameDebug.Veil() > 0, 'a jump through a gate should white the view out')
until GameDebug.Veil() == 0
until not GameDebug.Autopilot()
lua GameDebug.TimeScale(1)
lua assert(GameDebug.System() == _G.id, 'the autopilot stopped somewhere else')
lua assert(GameDebug.Destination() == -1, 'on arrival there is nowhere left to go')
lua print('arrived in', _G.far, 'which is', select(2, GameDebug.Galaxy()), 'space, security', (GameDebug.Security()))

# An order by hand takes the ship back from the autopilot.
lua assert(GameDebug.Destination(1) > 0)
lua assert(GameDebug.Autopilot(true))
frames 3
lua assert(GameDebug.Order('Stop'))
lua assert(not GameDebug.Autopilot(), 'an order by hand should switch the autopilot off')

# This is another faction's space, at its edge: the stations are its own.
lua GameDebug.TimeScale(16)
lua assert(GameDebug.Order('Dock', 'station'), 'dock was refused')
until GameDebug.Docked()
lua GameDebug.TimeScale(1)
frames 10
lua assert(select(2, GameDebug.Galaxy()) == 'Kethri', 'the far system in this galaxy should be the Kethri\'s')

echo galaxy scenario passed
quit
