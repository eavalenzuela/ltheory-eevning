# What is fitted to a hull of our own design stands on its hardpoints
# (script/check runs this):
#
#   bin/launch game --no-present --fixed-step 60 --script test/scenario/arms.txt
#
# The other scenarios fly the starter courier, which is one of the
# engine's ships and carries nothing to see. Here the pilot is put in a
# Vessari cruiser with the fit its design is drawn with: its weapons
# must be on its hardpoints when it undocks (each on both mounts of its
# hardpoint, which are a pair on opposite sides of the hull), all of
# them must fire, and of its two turrets the one that faces the target
# must come to point at it.
#
# And no hull of any faction has a hardpoint on a surface too small for
# what stands on it: the whole foot of every weapon, at its real size
# beside its hull, has hull under it (the project owner's rule).

wait 2
lua assert(GameDebug.Fly('cruiser.ouyou'), 'the Ouyou could not be flown')
lua assert(GameDebug.Hull() == 'cruiser.ouyou', 'the pilot is not in the Ouyou')
lua assert(GameDebug.Fitted('artillery.medium') == 1, 'the Ouyou did not take its gun')
lua assert(GameDebug.Fitted('launcher.heavy') == 3, 'the Ouyou did not take its launchers')
lua assert(GameDebug.Order('Undock'), 'undock was refused')
until GameDebug.InSpace()
frames 5
lua assert(GameDebug.Arms() == 8, 'four weapons, each on both mounts of its hardpoint, should stand on the Ouyou')

lua GameDebug.TimeScale(4)
lua assert(GameDebug.Order('WarpTo', 'belt'), 'warp to the belt was refused')
until GameDebug.InWarp()
until GameDebug.Idle()
until GameDebug.Pirates() > 0
until GameDebug.Distance('Pirate') < 18000
lua assert(GameDebug.Lock('Pirate'), 'the lock was refused')
until GameDebug.Locked() > 0
lua assert(GameDebug.Fire() == 4, 'all four weapons should open fire')
lua GameDebug.TimeScale(1)

# A medium gun swings at a little under a radian a second: five seconds
# bring it round from anywhere.
wait 5
lua print('the turret is', select(2, GameDebug.Arms()), 'radians off its target')
lua assert(select(2, GameDebug.Arms()) >= 0, 'the turret has nothing to point at')
lua assert(select(2, GameDebug.Arms()) < 0.05, 'the turret that faces the target is not pointing at it')

# Every hull there is a design for: nothing hangs off into space.
lua assert(GameDebug.Hulls() == 0, 'a hull has a weapon that does not stand wholly on it (each is printed above)')

echo arms scenario passed
quit
