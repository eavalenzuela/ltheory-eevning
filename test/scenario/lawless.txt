# Out of secure space, in the game app (script/check runs this):
#
#   bin/launch game --no-present --fixed-step 60 --script test/scenario/lawless.txt
#
# Space is safest where a pilot starts. Through the gate order is thin,
# and a belt brings a pack of raiders, not a scout. Through the next
# nobody keeps order at all: the packs have cruisers in them, and the
# belts have the rare ores. The pirates fly hulls of their own, which
# the scene must draw as those hulls with their weapons on them. A new
# pilot's starter frigate is no match for any of it. In low security
# the pack locks it and holds it with a warp disruptor before it can
# align, and it is lost (the owner's ruling, 2026-11-19: caught is
# fine, low security is meant to be deadly to a starter frigate). The
# pilot goes on in their capsule, which the pirates leave be, docks in
# lawless space, and takes another frigate out to look at a belt.
#
# covers: Core/Tactical/Pirates Core/Describe/Galaxy Core/Assets/Standings

wait 2
# (the ship the pilot begins in, whatever it is)
lua _G.mine = GameDebug.Hull()
# (the sound, at no loudness, so that what is heard of a fight is run: missiles land on the pilot here)
lua GameDebug.Sound(true, 0)
lua assert(GameDebug.Docked(), 'a new game starts docked')
lua assert(select(2, GameDebug.Security()) == 'secure', 'a pilot starts where order is kept')
lua assert(GameDebug.Security() == 1)
until GameDebug.Press('UNDOCK')
until GameDebug.InSpace()
lua GameDebug.TimeScale(8)

# Low security.
lua assert(GameDebug.Order('Jump', 'Kesis'), 'the jump out of secure space was refused')
until GameDebug.System() == 2
until GameDebug.Idle()
lua assert(select(2, GameDebug.Security()) == 'low', 'the system next door should be low security')
lua assert(GameDebug.Order('WarpTo', 'belt'), 'warp to the belt was refused')
until GameDebug.InWarp()
until GameDebug.Idle()
until GameDebug.Pirates() >= 2
frames 5
lua assert(GameDebug.Pirates() >= 2, 'a belt in low security should bring a pack')
lua assert(select(2, GameDebug.Pirates()) == GameDebug.Pirates(), 'a pirate is not drawn as a hull of the pirates\' own with its weapons on it')
lua print('low security:', GameDebug.Pirates(), 'pirates,', GameDebug.Distance('Pirate'), 'm off')

# Lawless space. The pack holds the frigate and destroys it before it
# can get away. In the capsule, which nobody holds, the pilot goes on.
lua assert(GameDebug.Order('Jump', 'K-PNDZ'), 'the jump into lawless space was refused')
patience 200
until GameDebug.System() == 3 or GameDebug.Hull() ~= _G.mine
lua assert(GameDebug.System() ~= 3, 'the starter frigate should not have got out of the pack\'s reach')
until GameDebug.Order('Jump', 'K-PNDZ')
until GameDebug.System() == 3
until GameDebug.Idle()
lua assert(select(2, GameDebug.Security()) == 'lawless', 'two jumps out should be lawless')
lua assert(GameDebug.Security() == 0)

# Nobody comes for a capsule: it docks at the one station there is
# here, and the pilot takes another frigate of the kind they began in
# (a cheat), to look at a belt.
lua assert(GameDebug.Order('Dock', 'station'), 'dock was refused')
until GameDebug.Docked()
lua assert(GameDebug.Fly(_G.mine), 'no frigate to be had')
until GameDebug.Press('UNDOCK')
until GameDebug.InSpace()
lua assert(GameDebug.Order('WarpTo', 'belt'), 'warp to the belt was refused')
until GameDebug.InWarp()
until GameDebug.Idle()
until GameDebug.Pirates() >= 3
frames 5
lua assert(select(2, GameDebug.Pirates()) == GameDebug.Pirates(), 'a pirate in lawless space is not drawn as its own hull')
lua assert(GameDebug.Distance('Cobaltite') < 1e9 or GameDebug.Distance('Iridite') < 1e9, 'a lawless belt should have rare ore')

# (whether it gets away from here is as it falls: lawless space is no
# place for a starter frigate either)

echo lawless scenario passed
quit
