# A hired pilot sent to buy, in the game app (script/check runs this):
#
#   bin/launch game --no-present --fixed-step 60 --script test/scenario/errand.txt
#
# Money and a hauler are given, and a pilot is hired and given the
# hauler. On the Market tab, at fuel blocks, the line under the buttons
# says where else they are to be had and what they cost there, and its
# button sends the pilot for as many as the quantity says. The player
# sits docked. The pilot goes to the other market, buys them, brings
# them back to the hangar here, and waits. What they cost has gone out
# of the wallet, and no more than was allowed.
#
# Then the same as a standing order: with twelve hundred asked for, the
# line's Keep has the pilot keep that many in the hangar. They go for
# the two hundred that are missing, and then wait, on the order still.
#
# covers: Core/Tactical/Fleet Core/Assets/Market

wait 2
lua assert(GameDebug.Docked(), 'a new game starts docked')
lua GameDebug.Peaceful(true)
lua GameDebug.Grant(1000000)
lua assert(GameDebug.Spare('industrial.hauler'), 'no hauler')
lua assert(GameDebug.Market('fuel.block'), 'no fuel on the market')
frames 3
until GameDebug.Press('Quantity:1000')
frames 3

# Somewhere else has them, and with nobody hired there is nobody to send.
lua assert(GameDebug.Sources('fuel.block', 1000) >= 1, 'no other market has fuel')
lua assert(select(3, GameDebug.Sources('fuel.block', 1000)) == 0, 'the best market should be the other one of this system')
lua assert(not GameDebug.CanPress('Fetch'), 'with nobody hired there is nobody to send')

until GameDebug.Press('Crew')
until GameDebug.Tab() == 'Crew'
until GameDebug.Press('Hire:1')
until GameDebug.Fleet() == 1
until GameDebug.Press('Give:1:1')
until select(3, GameDebug.Fleet()) == 1
until GameDebug.Press('Market')
until GameDebug.Tab() == 'Market'
frames 3

# Sent: out they go, with a thousand to buy.
lua _G.before = GameDebug.Wallet()
lua _G.each = select(4, GameDebug.Sources('fuel.block', 1000))
until GameDebug.Press('Fetch')
until GameDebug.Crew(1) == 'going to buy'
lua assert(select(9, GameDebug.Crew(1)) == 1000, 'a thousand should be still to buy')
lua assert(GameDebug.Hangar('fuel.block') == 0)
frames 3
lua assert(not GameDebug.CanPress('Fetch'), 'one out on an errand is not sent on another')

# Bought there, brought here, and paid for.
lua GameDebug.TimeScale(16)
until GameDebug.Hangar('fuel.block') == 1000
until GameDebug.Crew(1) == 'waiting'
lua GameDebug.TimeScale(1)
lua assert(select(3, GameDebug.Fleet()) == 1, 'the hauler should be docked again')
lua assert(select(2, GameDebug.Crew(1)) == 1000, 'a thousand units should be counted as brought in')
lua _G.spent = _G.before - GameDebug.Wallet()
lua assert(_G.spent > 0 and _G.spent <= math.ceil(_G.each * 1.1) * 1000, 'what was spent should be what a thousand cost there, and no more than allowed')
lua assert(math.abs(_G.spent - _G.each * 1000) <= 1, 'with nobody else buying, they cost what was quoted')
frames 3
lua assert(GameDebug.CanPress('Fetch'), 'back and waiting, they can be sent again')

# Twelve hundred to be kept here: two hundred are missing, and are fetched.
lua GameDebug.Enter('Quantity', 1200)
frames 3
until GameDebug.Press('Fetch:keep')
until GameDebug.Crew(1) == 'going to buy'
lua assert(select(9, GameDebug.Crew(1)) == 200, 'two hundred should be still to buy')
lua GameDebug.TimeScale(16)
until GameDebug.Hangar('fuel.block') == 1200
until GameDebug.Crew(1) == 'keeping stock'
lua GameDebug.TimeScale(1)
frames 3
lua assert(not GameDebug.CanPress('Fetch') and not GameDebug.CanPress('Fetch:keep'), 'one who keeps stock here is on that errand still')
echo errand scenario passed
quit
