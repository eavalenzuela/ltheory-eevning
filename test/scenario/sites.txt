# Sites on nobody's chart, in the game app (script/check runs this):
#
#   bin/launch game --no-present --fixed-step 60 --script test/scenario/sites.txt
#
# The first system has two signals the ship's scanner can pick up, in
# this game a vapour field and a hidden belt. Neither is on the
# overview. Three scans resolve them. Then they are places like any
# other: warp to the first, the vapour field, and its clouds are there
# to draw off (gas.txt does that). A probe launcher is fitted before the
# ship leaves (the module and the skill for it are given), and there
# the probes are put out at once, formed up smaller from the system
# map, and called in again.
#
# covers: Core/Tactical/Sites Core/Tactical/Vapour Game/UI/Scanner Game/UI/SystemMap

wait 2
lua GameDebug.Peaceful(true)
# (the sound, at no loudness: a place has its air, and a script can ask which)
lua GameDebug.Sound(true, 0)
wait 1
lua assert(GameDebug.Air() == 'chatter.busy', 'a homeworld station should have a good deal of chatter')
lua assert(GameDebug.Give('scanner.probe', 1))
lua GameDebug.Train('astrometrics', 1)
frames 3
until GameDebug.Press('Fit:scanner.probe')
until GameDebug.Fitted('scanner.probe') == 1
until GameDebug.Press('UNDOCK')
until GameDebug.InSpace()
lua assert(GameDebug.Order('Stop'))
wait 1
lua assert(GameDebug.Air() == 'chatter.busy', 'outside the station its chatter should still be heard')
lua assert(GameDebug.Count('site') == 0, 'an unfound site should not be on the overview')
until GameDebug.Press('Scanner')
frames 3
lua assert(GameDebug.Sites() == 2, 'the scanner should show two signals here')
lua assert(select(2, GameDebug.Sites()) == 0, 'nothing should be found before a scan')
lua assert(not GameDebug.CanPress('SiteWarp:1'), 'an unresolved signal is not somewhere to warp to')

# Three scans. The scanner takes a few seconds each time.
until GameDebug.Press('Scan')
until not GameDebug.CanPress('Scan')
lua assert(select(2, GameDebug.Sites()) == 0, 'a hidden belt or a vapour field should take more than one scan')
until GameDebug.CanPress('Scan')
until GameDebug.Press('Scan')
until not GameDebug.CanPress('Scan')
until GameDebug.CanPress('Scan')
until GameDebug.Press('Scan')
until select(2, GameDebug.Sites()) == 2
lua assert(GameDebug.Count('site') == 2, 'a found site should be on the overview')
lua assert(select(3, GameDebug.Sites('ore')) ~= '', 'one of the sites here should be a hidden belt')
lua assert(select(3, GameDebug.Sites('gas')) ~= '', 'and the other a vapour field')

# To the first of them, where there are clouds nobody else is drawing off.
frames 3
lua GameDebug.TimeScale(16)
until GameDebug.Press('SiteWarp:1')
until GameDebug.InWarp()
until GameDebug.Air() == ''
until GameDebug.Idle()
lua GameDebug.TimeScale(1)
wait 1
lua assert(GameDebug.Air() == 'gas', 'a vapour field should have its own air')
lua assert(select(2, GameDebug.Air()) <= 2, 'the chatter left behind should be gone, or going')
lua assert(GameDebug.Count('asteroid') >= 4, 'a vapour field should have its clouds')
lua assert(GameDebug.Distance('Vapour Field') < 1000, 'the warp did not end at the site')

# Probes: all that may be out are put out at once, and fly to a box round the ship.
lua assert(not GameDebug.CanPress('Probe:launch'), 'a probe has to be in the hold')
lua assert(GameDebug.Give('probe.scan', 10))
frames 3
until GameDebug.Press('Probe:launch')
until GameDebug.Probes() == 8
lua assert(select(2, GameDebug.Probes()) == 2, 'two probes should be left in the hold')
until select(3, GameDebug.Probes()) == 0
# On the system map: their reach two steps smaller, formed up again to it, and a scan.
until GameDebug.Press('Dock:systemmap')
frames 3
lua assert(select(4, GameDebug.Probes()) == 8, 'a new pilot\'s probes should reach 8 AU')
until GameDebug.Press('SystemMap:reach-')
until select(4, GameDebug.Probes()) == 4
until GameDebug.Press('SystemMap:reach-')
until select(4, GameDebug.Probes()) == 2
until GameDebug.Press('SystemMap:form')
# (a step or two: what they move is less than they go in a tick)
wait 2
until select(3, GameDebug.Probes()) == 0
until GameDebug.CanPress('SystemMap:scan')
until GameDebug.Press('SystemMap:scan')
until not GameDebug.CanPress('SystemMap:scan')
frames 3
until GameDebug.Press('SystemMap:recall')
until GameDebug.Probes() == 0
lua assert(select(2, GameDebug.Probes()) == 10, 'the probes called in should be in the hold again')
until GameDebug.Press('Dock:systemmap')

# Where people live there is comms chatter, whoever they are: near a
# planet people live on as at a station, and none by a planet nobody
# lives on.
lua _G.world, _G.waste = GameDebug.World(true), GameDebug.World(false)
lua assert(_G.world ~= '' and _G.waste ~= '', 'the system should have a planet people live on, and one nobody does')
lua GameDebug.TimeScale(16)
lua assert(GameDebug.Order('WarpTo', _G.world), 'warp to the planet was refused')
until GameDebug.InWarp()
until GameDebug.Idle()
lua GameDebug.TimeScale(1)
wait 1
lua assert(GameDebug.Air() == 'chatter.busy', 'near a homeworld planet people live on there should be a good deal of chatter, not ' .. GameDebug.Air())
lua GameDebug.TimeScale(16)
lua assert(GameDebug.Order('WarpTo', _G.waste), 'warp to the other planet was refused')
until GameDebug.InWarp()
until GameDebug.Idle()
lua GameDebug.TimeScale(1)
wait 1
lua assert(GameDebug.Air() == '', 'by a planet nobody lives on there is no chatter, not ' .. GameDebug.Air())

echo sites scenario passed
quit
