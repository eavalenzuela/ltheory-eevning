# A corporation founded and put to work, in the game app (script/check
# runs this):
#
#   bin/launch game --no-present --fixed-step 60 --script test/scenario/corporation.txt
#
# The Wallet tab has a side for the pilot's corporation. There is none
# yet: founding one is refused without a name, and without the fee. With
# a name typed and the fee in the wallet it is founded. Credits are paid
# into its account and some taken out again, which its books count as
# neither earned nor spent. Then members are asked into the fleet on the
# Crew window (since 2026-11-21 nobody is hired): nobody pays for it, and
# the books, looked at by head, have nothing under any.
#
# covers: Core/Assets/Corporation Core/Assets/Wallet Game/UI/Crew Game/UI/Wallet

wait 2
lua assert(GameDebug.Docked(), 'a new game starts docked')
until GameDebug.Press('Wallet')
until GameDebug.Tab() == 'Wallet'
frames 3
until GameDebug.Press('Wallet:corporation')
frames 3
lua assert(GameDebug.Corporation() == '', 'nobody has founded anything yet')
lua assert(select(5, GameDebug.Corporation()) == 0, 'there is no fleet without a corporation')
lua assert(not GameDebug.CanPress('Corporation:found'), 'a corporation with no name')
lua GameDebug.Type('Corporation:name', 'Merear Carriers')
frames 3
lua assert(not GameDebug.CanPress('Corporation:found'), 'a corporation for the 5,000 credits a pilot starts with')
lua GameDebug.Credits(900000)
frames 3
until GameDebug.Press('Corporation:found')
until GameDebug.Corporation() == 'Merear Carriers'
lua assert(GameDebug.Wallet() == 650000, 'founding should cost 250,000')
lua assert(select(5, GameDebug.Corporation()) == 20, 'a fleet of ten, and ten more for Leadership 1')
frames 3

# Paid in, and some of it taken out again.
lua assert(not GameDebug.CanPress('Corporation:deposit'), 'nothing has been said to pay in')
lua GameDebug.Enter('Corporation:amount', 600000)
frames 3
until GameDebug.Press('Corporation:deposit')
until select(2, GameDebug.Corporation()) == 600000
lua assert(GameDebug.Wallet() == 50000, 'what is paid in comes out of the wallet')
lua GameDebug.Enter('Corporation:amount', 100000)
frames 3
until GameDebug.Press('Corporation:withdraw')
until select(2, GameDebug.Corporation()) == 500000
lua assert(GameDebug.Wallet() == 150000, 'what is taken out goes into the wallet')
lua assert(select(3, GameDebug.Corporation()) == 0 and select(4, GameDebug.Corporation()) == 0, 'its founder paying in is neither earning nor spending')

# Members asked into the fleet (since 2026-11-21 nobody is hired): nobody
# pays to sign them, and nothing is on the books for it.
lua assert(GameDebug.Recruit(3, 'fighter', 2) == 3, 'no members to ask in')
until GameDebug.Press('Crew')
until GameDebug.Tab() == 'Crew'
frames 3
until GameDebug.Press('Invite:1')
until GameDebug.Fleet() == 1
frames 3
until GameDebug.Press('Invite:1')
until GameDebug.Fleet() == 2
frames 3
until GameDebug.Press('Invite:1')
until GameDebug.Fleet() == 3
lua assert(GameDebug.Wallet() == 150000 and select(2, GameDebug.Corporation()) == 500000, 'nobody should have paid for them')

# Its books, by head: nothing has been spent, and the pilot has no tower.
until GameDebug.Press('Wallet')
until GameDebug.Tab() == 'Wallet'
frames 3
until GameDebug.Press('Corporation:books')
frames 3
lua assert(GameDebug.CanPress('Corporation:journal'), 'the books should have their way back to the journal')
lua assert(GameDebug.Books() == 0, 'nothing should be on the books')
lua assert(select(4, GameDebug.Books()) == 0, 'no tower is on the books')
until GameDebug.Press('Corporation:journal')
frames 3

# Its hangar here: empty, until the pilot gives it some of the rounds
# the starter kit came with. They are taken back, given again, and sold
# for the corporation's account.
until GameDebug.Press('Corporation:hangar')
frames 3
lua assert(select(2, GameDebug.CorpHangar('ammo.emp')) == 0, 'the corporation should have nothing here yet')
lua _G.rounds = GameDebug.Hangar('ammo.emp')
lua assert(_G.rounds > 0, 'the starter kit should have rounds in it')
until GameDebug.Press('ToCorp:ammo.emp')
until GameDebug.CorpHangar('ammo.emp') == _G.rounds
lua assert(GameDebug.Hangar('ammo.emp') == 0, 'the rounds should have left the hangar of the pilot')
frames 3
until GameDebug.Press('FromCorp:ammo.emp')
until GameDebug.Hangar('ammo.emp') == _G.rounds
frames 3
until GameDebug.Press('ToCorp:ammo.emp')
until GameDebug.CorpHangar('ammo.emp') == _G.rounds
frames 3
lua _G.account = select(2, GameDebug.Corporation())
lua _G.own = GameDebug.Wallet()
until GameDebug.Press('SellCorp:ammo.emp')
until GameDebug.CorpHangar('ammo.emp') == 0
lua assert(select(2, GameDebug.Corporation()) > _G.account, 'the sale should have been paid to the corporation')
lua assert(GameDebug.Wallet() == _G.own, 'and none of it to the pilot')

echo corporation scenario passed
quit
