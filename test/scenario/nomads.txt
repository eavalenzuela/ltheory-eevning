# The Kethri's colony ships, in the game app (script/check runs this):
#
#   bin/launch game --no-present --fixed-step 60 --script test/scenario/nomads.txt
#
# The Kethri have no stations: where another people has one, a colony
# ship lies, ten kilometres long, with everything aboard. A galaxy of
# forty has Kethri systems in it. The pilot is put aboard the first
# colony ship there is (a cheat: it is a long way), finds every service
# a station gives, leaves it, sees it for what it is, and moors again.

wait 2
lua GameDebug.NewGame(3, 40)
wait 2
lua assert(GameDebug.Docked(), 'a new game starts docked')
lua _G.ship = GameDebug.Moor('kethri')
lua assert(string.find(_G.ship, 'Colony ship') ~= nil, 'the Kethri station should be a colony ship, not ' .. _G.ship)
wait 2
lua assert(GameDebug.Docked())
shot nomads-aboard

# Everything is aboard: the market, fitting, and a hull to buy.
until GameDebug.Press('Market')
until GameDebug.Tab() == 'Market'
until GameDebug.Press('Fitting')
until GameDebug.Tab() == 'Fitting'
shot nomads-fitting

# Out of it: it is there, it is what the overview says it is, and it is
# as big as it should be.
until GameDebug.Press('UNDOCK')
until GameDebug.InSpace()
wait 3
lua assert(GameDebug.Order('Stop'))
lua assert(GameDebug.Select(_G.ship), 'the colony ship should be on the overview')
lua assert(GameDebug.LookAt(_G.ship, 60))
wait 2
shot nomads-ship
lua GameDebug.Zoom(300, nil, 0.5)
wait 2
shot nomads-ship-above

# And back aboard.
lua assert(GameDebug.Order('Dock', _G.ship))
patience 240
until GameDebug.Docked()

echo nomads scenario passed
quit
